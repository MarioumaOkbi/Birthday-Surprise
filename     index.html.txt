<script>
function startExperience() {
  document.getElementById("entry").style.display = "none";
  document.getElementById("videoPopup").style.display = "flex";
  document.getElementById("music").play();
}

document.getElementById("introVideo").onended = function() {
  document.getElementById("videoPopup").style.display = "none";
};

setInterval(() => {
  let heart = document.createElement("div");
  heart.className = "heart";
  heart.innerHTML = "❤️";
  heart.style.left = Math.random() * 100 + "vw";
  document.body.appendChild(heart);
  setTimeout(() => heart.remove(), 6000);
}, 500);

function chooseGift() {
  document.getElementById("mixMatch").classList.remove("hidden");
  document.getElementById("mixMatch").scrollIntoView({behavior:"smooth"});
}

let riddles = [
  { q: "RIDDLE A (edit later)", a: "a" },
  { q: "RIDDLE B (edit later)", a: "b" },
  { q: "RIDDLE C (edit later)", a: "c" },
  { q: "RIDDLE D (edit later)", a: "d" },
  { q: "RIDDLE E (edit later)", a: "e" }
];

let currentIndex = 0;

function openRiddle(index) {
  currentIndex = index;
  document.getElementById("riddleText").innerText = riddles[index].q;
  document.getElementById("riddleBox").style.display = "flex";
}

function checkAnswer() {
  let input = document.getElementById("answerInput").value.toLowerCase();

  if (input === riddles[currentIndex].a) {
    document.getElementsByClassName("card")[currentIndex + 3].innerText =
      riddles[currentIndex].a.toUpperCase();
    document.getElementById("riddleBox").style.display = "none";
  } else {
    alert("Try again 😏❤️");
  }
}
